```bash
# 접근 정책(RLS)을 실제 PostgreSQL 에서 (빈 DB 필요, psycopg 설치)

SUPABASE_URL=... SUPABASE_KEY=... python scripts/check_supabase.py --email-a 시험1 --email-b 시험2
```
